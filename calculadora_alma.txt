=== COLUMNAS HOJA CALCULO (entradas y salidas) ===
A=Fila, B=Ref, C=Tipo artículo, D=Descripción, E=Ref tejido directo,
F=REF tejido (clave), G=Código color, H=Dobladillo, I=Ancho directo,
J=Ancho uso(cm), K=Largo uso(cm), L=Dobladillo(cm), M=Medida corte,
N=Encogimiento(1=Sí), O=Color nombre, P=Precio unitario, Q=Cantidad, R=Total

=== FÓRMULAS CALCULO (filas 2-10 ejemplo, luego 11-50 igual) ===
  CALCULO!M11: =IFERROR(CALCULOS!G11&" * "&CALCULOS!H11,"")
  CALCULO!O11: =IFERROR(VLOOKUP(F11&TEXT(G11,"0"),COLORES!$E$2:$G$4400,3,0),"")
  CALCULO!P11: =IFERROR(CALCULOS!I11+CALCULOS!J11+CALCULOS!K11,"")
  CALCULO!R11: =IFERROR(P11*Q11,"")
  CALCULO!M12: =IFERROR(CALCULOS!G12&" * "&CALCULOS!H12,"")
  CALCULO!O12: =IFERROR(VLOOKUP(F12&TEXT(G12,"0"),COLORES!$E$2:$G$4400,3,0),"")
  CALCULO!P12: =IFERROR(CALCULOS!I12+CALCULOS!J12+CALCULOS!K12,"")
  CALCULO!R12: =IFERROR(P12*Q12,"")

=== FÓRMULAS CALCULOS (intermedias) ===
  CALCULOS!A2: =MATCH(CALCULO!E2,ARTICULOS!F5:F200,0)
  CALCULOS!B2: =INDEX(ARTICULOS!D5:D200,A2)&INDEX(ARTICULOS!E5:E200,A2)
  CALCULOS!C2: =B2&CALCULO!G2
  CALCULOS!D2: =MATCH(C2,COLORES!E2:E4400,0)
  CALCULOS!E2: =INDEX(COLORES!I2:I4400,D2)
  CALCULOS!F2: =INDEX(COLORES!H2:H4400,D2)
  CALCULOS!G2: =IF(CALCULO!I2>0,ROUNDUP(CALCULO!I2*(1+INDEX(ARTICULOS!H5:H200,MATCH(CALCULO!E2,ARTICULOS!F5:F200,0))),0),IF(CALCULO!N2=1,IF(CALCULO!H2=1,ROUNDUP((CALCULO!J2+CALCULO!L2*2+5)*(1+INDEX(ARTICULOS!H5:H200,MATCH(CALCULO!E2,ARTICULOS!F5:F200,0))),0),ROUNDUP((CALCULO!J2+CALCULO!L2*2+CALCULO!H2*4)*(1+INDEX(ARTICULOS!H5:H200,MATCH(CALCULO!E2,ARTICULOS!F5:F200,0))),0)),IF(CALCULO!H2=1,ROUNDUP(CALCULO!J2+CALCULO!L2*2+5,0),ROUNDUP(CALCULO!J2+CALCULO!L2*2+CALCULO!H2*4,0))))
  CALCULOS!H2: =IF(CALCULO!H2=1,ROUNDUP((CALCULO!K2+CALCULO!L2*2+5)*(1+INDEX(ARTICULOS!G5:G200,MATCH(CALCULO!E2,ARTICULOS!F5:F200,0))),0),ROUNDUP((CALCULO!K2+CALCULO!L2*2+CALCULO!H2*4)*(1+INDEX(ARTICULOS!G5:G200,MATCH(CALCULO!E2,ARTICULOS!F5:F200,0))),0))
  CALCULOS!I2: =ROUND(F2*2*H2/100/INT(E2/(G2/100)),2)
  CALCULOS!J2: =IF(CALCULO!H2=1,INDEX(CONFECCION!C15:AD30,MATCH(G2,CONFECCION!A15:A30,1),MATCH(H2,CONFECCION!C14:AD14,1)),2*INDEX(CONFECCION!C15:AD30,MATCH(G2,CONFECCION!A15:A30,1),MATCH(H2,CONFECCION!C14:AD14,1)))
  CALCULOS!K2: =IF(AND(CALCULO!C2="Servilleta",CALCULO!G2=58),ROUND((G2*2+H2*2)*CONFECCION!B32/100,2),IF(CALCULO!C2="Individual",ROUND((G2*2+H2*2)*CONFECCION!B32/100,2),0))
  CALCULOS!A3: =MATCH(CALCULO!E3,ARTICULOS!F6:F201,0)
  CALCULOS!B3: =INDEX(ARTICULOS!D6:D201,A3)&INDEX(ARTICULOS!E6:E201,A3)
  CALCULOS!C3: =B3&CALCULO!G3
  CALCULOS!D3: =MATCH(C3,COLORES!E3:E4401,0)
  CALCULOS!E3: =INDEX(COLORES!I3:I4401,D3)
  CALCULOS!F3: =INDEX(COLORES!H3:H4401,D3)
  CALCULOS!G3: =IF(CALCULO!I3>0,ROUNDUP(CALCULO!I3*(1+INDEX(ARTICULOS!H5:H200,MATCH(CALCULO!E3,ARTICULOS!F5:F200,0))),0),IF(CALCULO!N3=1,IF(CALCULO!H3=1,ROUNDUP((CALCULO!J3+CALCULO!L3*2+5)*(1+INDEX(ARTICULOS!H5:H200,MATCH(CALCULO!E3,ARTICULOS!F5:F200,0))),0),ROUNDUP((CALCULO!J3+CALCULO!L3*2+CALCULO!H3*4)*(1+INDEX(ARTICULOS!H5:H200,MATCH(CALCULO!E3,ARTICULOS!F5:F200,0))),0)),IF(CALCULO!H3=1,ROUNDUP(CALCULO!J3+CALCULO!L3*2+5,0),ROUNDUP(CALCULO!J3+CALCULO!L3*2+CALCULO!H3*4,0))))
  CALCULOS!H3: =IF(CALCULO!H3=1,ROUNDUP((CALCULO!K3+CALCULO!L3*2+5)*(1+INDEX(ARTICULOS!G5:G200,MATCH(CALCULO!E3,ARTICULOS!F5:F200,0))),0),ROUNDUP((CALCULO!K3+CALCULO!L3*2+CALCULO!H3*4)*(1+INDEX(ARTICULOS!G5:G200,MATCH(CALCULO!E3,ARTICULOS!F5:F200,0))),0))
  CALCULOS!I3: =ROUND(F3*2*H3/100/INT(E3/(G3/100)),2)
  CALCULOS!J3: =IF(CALCULO!H3=1,INDEX(CONFECCION!C15:AD30,MATCH(G3,CONFECCION!A15:A30,1),MATCH(H3,CONFECCION!C14:AD14,1)),2*INDEX(CONFECCION!C15:AD30,MATCH(G3,CONFECCION!A15:A30,1),MATCH(H3,CONFECCION!C14:AD14,1)))
  CALCULOS!K3: =IF(AND(CALCULO!C3="Servilleta",CALCULO!G3=58),ROUND((G3*2+H3*2)*CONFECCION!B32/100,2),IF(CALCULO!C3="Individual",ROUND((G3*2+H3*2)*CONFECCION!B32/100,2),0))
  CALCULOS!A11: =IFERROR(IF(CALCULO!E11<>"",MATCH(CALCULO!E11,ARTICULOS!$F$5:$F$700,0),MATCH(CALCULO!F11&TEXT(CALCULO!G11,"0"),COLORES!$E$2:$E$4400,0)),"")
  CALCULOS!B11: =IFERROR(IF(CALCULO!E11<>"",INDEX(ARTICULOS!$D$5:$D$700,A11)&INDEX(ARTICULOS!$E$5:$E$700,A11),CALCULO!F11),"")
  CALCULOS!C11: =IFERROR(B11&TEXT(CALCULO!G11,"0"),"")
  CALCULOS!D11: =IFERROR(MATCH(C11,COLORES!$E$2:$E$4400,0),"")
  CALCULOS!E11: =IFERROR(INDEX(COLORES!$I$2:$I$4400,D11),"")
  CALCULOS!F11: =IFERROR(INDEX(COLORES!$H$2:$H$4400,D11),"")
  CALCULOS!G11: =IFERROR(IF(CALCULO!I11>0,ROUNDUP(CALCULO!I11*(1+IF(CALCULO!N11=1,IFERROR(INDEX(ARTICULOS!$H$5:$H$700,MATCH(INDEX(COLORES!$D$2:$D$4400,D11),ARTICULOS!$F$5:$F$700,0)),0),0)),0),IF(CALCULO!N11=1,IF(CALCULO!H11=1,ROUNDUP((CALCULO!J11+CALCULO!L11*2+5)*(1+IFERROR(INDEX(ARTICULOS!$H$5:$H$700,MATCH(INDEX(COLORES!$D$2:$D$4400,D11),ARTICULOS!$F$5:$F$700,0)),0)),0),ROUNDUP((CALCULO!J11+CALCULO!L11*2+CALCULO!H11*4)*(1+IFERROR(INDEX(ARTICULOS!$H$5:$H$700,MATCH(INDEX(COLORES!$D$2:$D$4400,D11),ARTICULOS!$F$5:$F$700,0)),0)),0)),IF(CALCULO!H11=1,ROUNDUP(CALCULO!J11+CALCULO!L11*2+5,0),ROUNDUP(CALCULO!J11+CALCULO!L11*2+CALCULO!H11*4,0)))),"")
  CALCULOS!H11: =IFERROR(IF(CALCULO!H11=1,ROUNDUP((CALCULO!K11+CALCULO!L11*2+5)*(1+IF(CALCULO!N11=1,IFERROR(INDEX(ARTICULOS!$G$5:$G$700,MATCH(INDEX(COLORES!$D$2:$D$4400,D11),ARTICULOS!$F$5:$F$700,0)),0),0)),0),ROUNDUP((CALCULO!K11+CALCULO!L11*2+CALCULO!H11*4)*(1+IF(CALCULO!N11=1,IFERROR(INDEX(ARTICULOS!$G$5:$G$700,MATCH(INDEX(COLORES!$D$2:$D$4400,D11),ARTICULOS!$F$5:$F$700,0)),0),0)),0)),"")
  CALCULOS!I11: =IFERROR(ROUND(F11*2*H11/100/IF(INT(E11/(G11/100))=0,1,INT(E11/(G11/100))),2),"")
  CALCULOS!J11: =IFERROR(IF(CALCULO!C11="Lito",INDEX(CONFECCION!$C$15:$AD$30,MATCH(G11,CONFECCION!$A$15:$A$30,1),MATCH(H11,CONFECCION!$C$14:$AD$14,1)),IF(INT(E11/(G11/100))=0,2*INDEX(CONFECCION!$C$15:$AD$30,MATCH(G11,CONFECCION!$A$15:$A$30,1),MATCH(H11,CONFECCION!$C$14:$AD$14,1))*3,IF(CALCULO!H11=1,INDEX(CONFECCION!$C$15:$AD$30,MATCH(G11,CONFECCION!$A$15:$A$30,1),MATCH(H11,CONFECCION!$C$14:$AD$14,1)),2*INDEX(CONFECCION!$C$15:$AD$30,MATCH(G11,CONFECCION!$A$15:$A$30,1),MATCH(H11,CONFECCION!$C$14:$AD$14,1))))),"")
  CALCULOS!K11: =IFERROR(IF(OR(AND(CALCULO!C11="Servilleta",CALCULO!G11=58),CALCULO!C11="Individual"),ROUND((G11*2+H11*2)*CONFECCION!$B$32/100,2),0),"")
  CALCULOS!A12: =IFERROR(IF(CALCULO!E12<>"",MATCH(CALCULO!E12,ARTICULOS!$F$5:$F$700,0),MATCH(CALCULO!F12&TEXT(CALCULO!G12,"0"),COLORES!$E$2:$E$4400,0)),"")

  CALCULOS!E12: =IFERROR(INDEX(COLORES!$I$2:$I$4400,D12),"")

  CALCULOS!H12: =IFERROR(IF(CALCULO!H12=1,ROUNDUP((CALCULO!K12+CALCULO!L12*2+5)*(1+IF(CALCULO!N12=1,IFERROR(INDEX(ARTICULOS!$G$5:$G$700,MATCH(INDEX(COLORES!$D$2:$D$4400,D12),ARTICULOS!$F$5:$F$700,0)),0),0)),0),ROUNDUP((CALCULO!K12+CALCULO!L12*2+CALCULO!H12*4)*(1+IF(CALCULO!N12=1,IFERROR(INDEX(ARTICULOS!$G$5:$G$700,MATCH(INDEX(COLORES!$D$2:$D$4400,D12),ARTICULOS!$F$5:$F$700,0)),0),0)),0)),"")
  CALCULOS!I12: =IFERROR(ROUND(F12*2*H12/100/IF(INT(E12/(G12/100))=0,1,INT(E12/(G12/100))),2),"")
  CALCULOS!J12: =IFERROR(IF(CALCULO!C12="Lito",INDEX(CONFECCION!$C$15:$AD$30,MATCH(G12,CONFECCION!$A$15:$A$30,1),MATCH(H12,CONFECCION!$C$14:$AD$14,1)),IF(INT(E12/(G12/100))=0,2*INDEX(CONFECCION!$C$15:$AD$30,MATCH(G12,CONFECCION!$A$15:$A$30,1),MATCH(H12,CONFECCION!$C$14:$AD$14,1))*3,IF(CALCULO!H12=1,INDEX(CONFECCION!$C$15:$AD$30,MATCH(G12,CONFECCION!$A$15:$A$30,1),MATCH(H12,CONFECCION!$C$14:$AD$14,1)),2*INDEX(CONFECCION!$C$15:$AD$30,MATCH(G12,CONFECCION!$A$15:$A$30,1),MATCH(H12,CONFECCION!$C$14:$AD$14,1))))),"")
  CALCULOS!K12: =IFERROR(IF(OR(AND(CALCULO!C12="Servilleta",CALCULO!G12=58),CALCULO!C12="Individual"),ROUND((G12*2+H12*2)*CONFECCION!$B$32/100,2),0),"")

=== LÓGICA DE CÁLCULO ===

CALCULOS col A: posición del tejido/color en tablas auxiliares
CALCULOS col B: nombre tejido
CALCULOS col C: clave combinada tejido+color
CALCULOS col D: posición en tabla COLORES
CALCULOS col E: ancho del rollo (cm)
CALCULOS col F: precio €/metro
CALCULOS col G: ancho de corte con encogimiento (cm)
CALCULOS col H: largo de corte con encogimiento (cm)
CALCULOS col I: coste tejido = precio × 2 × largo_corte / piezas_por_ancho
CALCULOS col J: coste confección (tabla CONFECCION según medidas)
CALCULOS col K: festón si aplica (Servilleta color=58 o Individual)

CALCULO col M: medida corte "G × H"
CALCULO col O: nombre del color
CALCULO col P: precio unitario = I + J + K
CALCULO col R: total línea = P × Q

FESTÓN: ((G×2 + H×2) / 100) × 0.75 €/m
CONFECCIÓN:
  - Lito (código 7): tasa × 1
  - Sin empalme (piezas>0): tasa × 2
  - Con empalme (piezas=0): tasa × 2 × 3
  - Dobladillo festón (H=1): tasa × 1
ENCOGIMIENTO: busca en COLORES→ARTICULOS los % de largo y ancho
